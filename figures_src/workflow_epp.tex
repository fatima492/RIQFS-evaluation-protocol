% ------------------------------------------------------------------ procedure cards
\node[card=c1] (p1) at (\colA+22,-10.5) {\topbox{40mm}{%
  \iline{database-c1}{List programme types, criteria (outcomes or horizons) and their order of importance.}
  \iline{chart-bar-c1}{Take each summary the synthesis reports as a separate \textbf{channel}, e.g.\ a vote count \dots}
  \iline{ruler-measure-c1}{\dots\ and a mean effect, each with its uncertainty and clustering.}}};
\node[card=c2] (p2) at (\colB+22,-10.5) {\topbox{40mm}{%
  \iline{thumb-up-c2}{Turn each interval $[L,U]$ into \textbf{support} $\mu$,}
  \iline{thumb-down-c2}{\textbf{opposition} $\nu$}
  \iline{help-circle-c2}{and \textbf{uncommitted} evidence $1-\mu-\nu$.}
  \iline{adjustments-horizontal-c2}{\textbf{State the value choice:} what counts against a programme? Set reliabilities $\kappa$ and benchmark $M$.}}};
\node[card=c3] (p3) at (\colC+22,-10.5) {\topbox{40mm}{%
  \iline{chart-bar-c3}{Pool the channels into support $t_{ij}$ and opposition $f_{ij}$.}
  \iline{trophy-c3}{Rank by the weighted balance\\[1pt]
    $C_i=\tfrac12(1+S_i)$,\\ $S_i=\sum_j\beta_j(t_{ij}-f_{ij})$.}
  \iline{git-compare-c3}{Report where channels disagree \textbf{beside} the rank, not inside it.}}};
\node[card=c4] (p4) at (\colD+22,-10.5) {\topbox{40mm}{%
  \iline{chart-histogram-c4}{Bootstrap the published statistics (cluster-adjusted).}
  \iline{target-arrow-c4}{Map the whole weight space, with ordinal information.}
  \iline{arrows-shuffle-c4}{Vary the value choices and approximations.}
  \iline{coins-c4}{Compute break-even cost ratios.}}};
